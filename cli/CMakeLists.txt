cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(serial-port LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()

include(FetchContent)
FetchContent_Declare(asio
  URL https://codeload.github.com/chriskohlhoff/asio/tar.gz/42a93679dc4c8c5caf3d3082542f1bfa2438271d
  URL_HASH SHA256=51d0fba0828753266cde455ffde6239d624f5e383a630a4efed5eab64be00034)
FetchContent_Declare(json
  URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03
  URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(asio json)
find_package(Threads REQUIRED)
add_executable(serial-port main.cpp)
function(configure_serial_target target)
  target_include_directories(${target} PRIVATE ${asio_SOURCE_DIR}/asio/include)
  target_compile_definitions(${target} PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED)
  target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0601 NOMINMAX)
    target_link_libraries(${target} PRIVATE ws2_32 mswsock advapi32)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8)
  endif()
endfunction()
configure_serial_target(serial-port)
if(MINGW)
  target_link_options(serial-port PRIVATE -municode)
endif()
